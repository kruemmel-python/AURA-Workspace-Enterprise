cmake_minimum_required(VERSION 3.24)
project(AuraWorkspaceEnterprise VERSION 2.4.5 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(WIN32)
    add_executable(aura_workspace WIN32 aura_workspace.cpp core/enterprise_database.cpp)
    target_include_directories(aura_workspace PRIVATE "H:/KR-Compiler/KR-Compiler-Windows-x64-v4.3.1/toolchain/include")
    target_link_directories(aura_workspace PRIVATE "H:/KR-Compiler/KR-Compiler-Windows-x64-v4.3.1/toolchain/lib")
    target_link_libraries(aura_workspace PRIVATE sqlite3 user32 gdi32 comdlg32)
else()
    find_package(SQLite3 REQUIRED)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    add_library(aura_enterprise_database STATIC core/enterprise_database.cpp)
    target_link_libraries(aura_enterprise_database PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
    set_target_properties(aura_enterprise_database PROPERTIES POSITION_INDEPENDENT_CODE ON)
    find_program(AURA_RUSTC_EXECUTABLE rustc REQUIRED)
    set(AURA_RULES_LIB "${CMAKE_CURRENT_BINARY_DIR}/libaura_rules.a")
    add_custom_command(OUTPUT "${AURA_RULES_LIB}"
        COMMAND "${AURA_RUSTC_EXECUTABLE}" --edition=2021 --crate-type=staticlib -C relocation-model=pic -C opt-level=2
                "${CMAKE_CURRENT_SOURCE_DIR}/rules/src/lib.rs" -o "${AURA_RULES_LIB}"
        DEPENDS rules/src/lib.rs VERBATIM)
    add_custom_target(aura_linux_rules DEPENDS "${AURA_RULES_LIB}")
    add_library(aura_linux_local SHARED clients/linux/local_bridge.cpp)
    add_dependencies(aura_linux_local aura_linux_rules)
    target_link_libraries(aura_linux_local PRIVATE aura_enterprise_database "${AURA_RULES_LIB}" ${CMAKE_DL_LIBS} pthread m)

    include(CTest)
    if(BUILD_TESTING)
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/KR_BUILD")
        add_executable(enterprise_database_tests tests/enterprise_database_tests.cpp)
        target_link_libraries(enterprise_database_tests PRIVATE aura_enterprise_database)
        target_compile_options(enterprise_database_tests PRIVATE -UNDEBUG)
        add_test(NAME enterprise_database COMMAND enterprise_database_tests)
        add_executable(linux_database_compat_tests tests/linux_database_compat_tests.cpp)
        target_link_libraries(linux_database_compat_tests PRIVATE aura_enterprise_database)
        target_compile_options(linux_database_compat_tests PRIVATE -UNDEBUG)
        add_test(NAME linux_database_compat COMMAND linux_database_compat_tests)
    endif()
endif()
